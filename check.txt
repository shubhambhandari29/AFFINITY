SELECT
    p.AcctAffinityProgramKey AS ProgramId,
    p.ProgramName,
    a.AgentCode,
    a.LossRunReq
FROM dbo.tblAcctAffinityProgram p
JOIN dbo.tblAffinityAgents a
    ON a.ProgramName = p.ProgramName
WHERE a.LossRunReq = 'Yes'
  AND NULLIF(LTRIM(RTRIM(a.AgentCode)), '') IS NOT NULL
ORDER BY p.ProgramName, a.AgentCode;
