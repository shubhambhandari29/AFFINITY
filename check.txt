-- Confirm every HCM date-related column’s exact type
SELECT
    COLUMN_NAME,
    DATA_TYPE,
    DATETIME_PRECISION,
    IS_NULLABLE,
    COLUMN_DEFAULT
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_NAME = 'tblHcmAccount'
  AND (
      COLUMN_NAME LIKE '%Date%'
      OR COLUMN_NAME LIKE '%Dt%'
  )
ORDER BY ORDINAL_POSITION;



-- Check triggers specific to HCM
SELECT
    tr.name AS TriggerName,
    OBJECT_DEFINITION(tr.object_id) AS TriggerDefinition
FROM sys.triggers tr
WHERE tr.parent_id = OBJECT_ID('tblHcmAccount');
