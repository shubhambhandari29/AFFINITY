SET XACT_ABORT ON;

BEGIN TRANSACTION;

-- 1. tblHcmAccount
-- Use the existing UpdateDateTime as the best available historical value
-- before making InsertDateTime mandatory.
UPDATE [dbo].[tblHcmAccount]
SET [InsertDateTime] = [UpdateDateTime]
WHERE [InsertDateTime] IS NULL;

ALTER TABLE [dbo].[tblHcmAccount]
ALTER COLUMN [InsertDateTime] [datetimeoffset](7) NOT NULL;

IF NOT EXISTS (
	SELECT 1
	FROM sys.indexes
	WHERE [object_id] = OBJECT_ID(N'[dbo].[tblHcmAccount]')
		AND [name] = N'IX_tblHcmAccount_Stage_IsSubmitted'
)
BEGIN
	CREATE NONCLUSTERED INDEX [IX_tblHcmAccount_Stage_IsSubmitted]
	ON [dbo].[tblHcmAccount] ([Stage], [IsSubmitted])
	INCLUDE ([CustomerNum], [CustomerName], [OnBoardDate], [AcctStatus]);
END;


-- 2. tblHcmAccountAssociations
-- Add the columns as nullable first so the change works when rows already exist.
ALTER TABLE [dbo].[tblHcmAccountAssociations]
ADD
	[InsertDateTime] [datetimeoffset](7) NULL,
	[UpdateDateTime] [datetimeoffset](7) NULL;

-- There is no historical audit date in this table, so existing association
-- rows receive the time at which this PRE-PROD migration is executed.
UPDATE [dbo].[tblHcmAccountAssociations]
SET
	[InsertDateTime] = SYSDATETIMEOFFSET(),
	[UpdateDateTime] = SYSDATETIMEOFFSET()
WHERE [InsertDateTime] IS NULL
	OR [UpdateDateTime] IS NULL;

ALTER TABLE [dbo].[tblHcmAccountAssociations]
ALTER COLUMN [InsertDateTime] [datetimeoffset](7) NOT NULL;

ALTER TABLE [dbo].[tblHcmAccountAssociations]
ALTER COLUMN [UpdateDateTime] [datetimeoffset](7) NOT NULL;

IF NOT EXISTS (
	SELECT 1
	FROM sys.indexes
	WHERE [object_id] = OBJECT_ID(N'[dbo].[tblHcmAccountAssociations]')
		AND [name] = N'IX_tblHcmAccountAssociations_Parent_Associated'
)
BEGIN
	CREATE NONCLUSTERED INDEX [IX_tblHcmAccountAssociations_Parent_Associated]
	ON [dbo].[tblHcmAccountAssociations] ([ParentAccount], [AssociatedAccount]);
END;


-- 3. tblHcmOnlyUsers
-- Use the existing UpdateDateTime as the best available historical value
-- before making InsertDateTime mandatory.
UPDATE [dbo].[tblHcmOnlyUsers]
SET [InsertDateTime] = [UpdateDateTime]
WHERE [InsertDateTime] IS NULL;

ALTER TABLE [dbo].[tblHcmOnlyUsers]
ALTER COLUMN [InsertDateTime] [datetimeoffset](7) NOT NULL;

IF NOT EXISTS (
	SELECT 1
	FROM sys.indexes
	WHERE [object_id] = OBJECT_ID(N'[dbo].[tblHcmOnlyUsers]')
		AND [name] = N'IX_tblHcmOnlyUsers_CustNum'
)
BEGIN
	CREATE NONCLUSTERED INDEX [IX_tblHcmOnlyUsers_CustNum]
	ON [dbo].[tblHcmOnlyUsers] ([CustNum]);
END;

COMMIT TRANSACTION;
